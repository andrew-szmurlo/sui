4//! Scan-then-live subscription driver: stream the items matching a filter, in checkpoint order,
5//! resuming from a cursor or checkpoint, in two phases:
6//!
7//! 1. Backfill ([`backfill`]): scan indexed data forward from the resume point to catch up, using
8//!    the same pagination path.
9//! 2. Live ([`live`]): read matching items from the checkpoint broadcast.
10//!
11//! Once the scan comes within `handoff_threshold` of the tip, the driver subscribes to the broadcast,
12//! so reading is handed off to the live phase at `handoff + 1` with no gap or duplicate. A coverage
13//! marker at each page's end lets the handoff advance even through stretches that match nothing.
14

56/// How long to wait before re-requesting when the scan has drained the indexer's current tip but not
57/// yet reached the handoff. The wait doubles while the ledger makes no progress, up to
58/// `BACKFILL_POLL_MAX_INTERVAL`.
59const BACKFILL_POLL_INTERVAL: Duration = Duration::from_millis(100);
60const BACKFILL_POLL_MAX_INTERVAL: Duration = Duration::from_secs(2);
61
62/// How long the backfill waits for the ledger to make progress before asking the client to
63/// reconnect, so a stalled ledger does not hold subscription slots indefinitely.
64const BACKFILL_POLL_DEADLINE: Duration = Duration::from_secs(180);

66/// A feed that can be streamed with the scan-then-live driver. One implementation per subscribable
67/// GraphQL type (transactions, events) supplies its cursor, filter, and matching, and the free
68/// functions here drive the shared machinery over it.
69pub(super) trait Subscribable {
70    /// The GraphQL node delivered in each edge.
71    type Item: OutputType + Send + 'static;
72    /// The opaque cursor minted for resumption.
73    type Cursor: CursorType
74        + Eq
75        + Clone
76        + Send
77        + Sync
78        + TryFrom<CursorToken, Error = anyhow::Error>
79        + 'static;
80    /// The filter selecting matching items.
81    type Filter: Clone + Send + Sync + 'static;
82    /// The raw scan payload one page yields before conversion to edges.
83    type ScanItem: Send + 'static;
84
85    /// Scan one page of matches over `cp_bounds`, starting after `page`'s cursor.
86    fn scan<'a>(
87        reader: &'a AlphaLedgerGrpcReader,
88        cp_bounds: RangeInclusive<u64>,
89        page: &'a Page<Self::Cursor>,
90        filter: &'a Self::Filter,
91    ) -> impl Future<Output = Result<StreamPage<Self::ScanItem>, RpcError>> + Send + 'a;
92
93    /// Build the GraphQL node from one scanned item's payload, constructing whatever scope the feed
94    /// resolves against. The driver mints the cursor from the item's position and assembles the edge.
95    fn build_node(
96        caches: &Arc<StreamedCaches>,
97        resolver_limits: &sui_package_resolver::Limits,
98        payload: &Self::ScanItem,
99    ) -> Result<Self::Item, RpcError>;
100
101    /// The filter-matching edges in one live checkpoint, in delivery order.
102    fn matching_edges(
103        checkpoint: &Arc<ProcessedCheckpoint>,
104        caches: &Arc<StreamedCaches>,
105        resolver_limits: &sui_package_resolver::Limits,
106        filter: &Self::Filter,
107    ) -> Result<Vec<Edge<String, Self::Item, EmptyFields>>, RpcError>;
108
109    /// The feed's name, used to label its lifecycle metrics.
110    fn subscription_type() -> &'static str;
111}

113/// One backfill scan output: either a matching edge, or a coverage marker whose `checkpoint` is the
114/// fully-scanned frontier (advances the handoff even on a match-less page).
115enum Scanned<I: OutputType> {
116    Match {
117        checkpoint: u64,
118        edge: Edge<String, I, EmptyFields>,
119    },
120    Covered {
121        checkpoint: u64,
122    },
123}
124
125/// Where the backfill scan hands off to the live broadcast. The scan delivers every match at or
126/// below the pinned handoff; live delivers everything above it.
127struct Handoff {
128    /// Pin once the scan frontier is within this many checkpoints of the streamed tip.
129    threshold: u64,
130    /// The checkpoint the client resumed after (from `after` and/or `afterCheckpoint`).
131    resume_from: Option<u64>,
132    /// The `after` cursor's checkpoint is a placeholder (see [`has_placeholder_checkpoint`]), so
133    /// `resume_from` is raised to the first scan output, which the ledger only returns once it is
134    /// past the cursor.
135    resume_unknown: bool,
136    pinned: Option<u64>,
137}
138
139/// What the backfill does with one scan output, as decided by [`Handoff::step`].
140enum ScanStep<I: OutputType> {
141    /// Deliver this match.
142    Deliver(Edge<String, I, EmptyFields>),
143    /// Keep scanning.
144    Continue,
145    /// The scan has covered the handoff; live takes over.
146    Done,
147}
148
149impl<I: OutputType> Scanned<I> {
150    /// The checkpoint this output sits at (the match's, or the covered frontier).
151    fn checkpoint(&self) -> u64 {
152        match self {
153            Scanned::Match { checkpoint, .. } | Scanned::Covered { checkpoint } => *checkpoint,
154        }
155    }
156}
157
158impl Handoff {
159    fn new(threshold: u64, resume_from: Option<u64>) -> Self {
160        Self {
161            threshold,
162            resume_from,
163            resume_unknown: false,
164            pinned: None,
165        }
166    }
167
168    /// Take the resume point from the first scan output, for an `after` cursor whose checkpoint is
169    /// a placeholder.
170    fn resume_from_first_output(mut self) -> Self {
171        self.resume_unknown = true;
172        self
173    }
174
175    /// Decide what to do with one scan output, given this replica's streamed `tip`.
176    ///
177    /// Pins first if the scan frontier is within the threshold of `tip`. `resubscribe` must
178    /// subscribe the live receiver and then return the streamed tip, so live sees every checkpoint
179    /// above the pin. A match past the handoff belongs to live, so the scan stops there.
180    fn step<I: OutputType>(
181        &mut self,
182        scanned: Scanned<I>,
183        tip: u64,
184        resubscribe: impl FnOnce() -> u64,
185    ) -> ScanStep<I> {
186        if std::mem::take(&mut self.resume_unknown) {
187            // The first output is past the cursor: a match sits in the cursor's checkpoint or a
188            // later one with no match in between, and a coverage marker's page ends in the
189            // checkpoint after the one it covers. Either way the scan stays responsible for it.
190            let floor = match &scanned {
191                Scanned::Match { checkpoint, .. } => *checkpoint,
192                Scanned::Covered { checkpoint } => checkpoint.saturating_add(1),
193            };
194            self.resume_from = Some(self.resume_from.map_or(floor, |from| from.max(floor)));
195        }
196
197        if self.pinned.is_none() && tip.saturating_sub(scanned.checkpoint()) <= self.threshold {
198            self.pin(resubscribe());
199        }
200
201        match scanned {
202            Scanned::Match { checkpoint, edge } => {
203                if self.pinned.is_none_or(|h| checkpoint <= h) {
204                    ScanStep::Deliver(edge)
205                } else {
206                    ScanStep::Done
207                }
208            }
209            Scanned::Covered { checkpoint } => {
210                if self.pinned.is_some_and(|h| checkpoint >= h) {
211                    ScanStep::Done
212                } else {
213                    ScanStep::Continue
214                }
215            }
216        }
217    }
218
219    /// Pin the handoff at the streamed `tip` (read after resubscribing), floored at the resume
220    /// point, which can be ahead of this replica's tip behind a load balancer. Flooring here keeps
221    /// the scan responsible for the matches after an `after` cursor in its own checkpoint.
222    fn pin(&mut self, tip: u64) {
223        self.pinned = Some(match self.resume_from {
224            Some(from) => tip.max(from),
225            None => tip,
226        });
227    }
228
229    /// The last checkpoint the scan was responsible for; live delivers from the next one.
230    fn pinned(&self) -> Option<u64> {
231        self.pinned
232    }
233}

235/// Subscribe to items matching `filter`: backfill from `resume` toward the tip, then follow live.
236/// The handoff is pinned mid-scan, once the scan frontier comes within `handoff_threshold` of the
237/// tip, so even a deep backfill catches up within one connection instead of lagging the receiver.
238#[allow(clippy::type_complexity)]
239pub(super) fn subscribe<S: Subscribable>(
240    reader: AlphaLedgerGrpcReader,
241    broadcast: Arc<SubscriptionBroadcast>,
242    caches: Arc<StreamedCaches>,
243    resolver_limits: sui_package_resolver::Limits,
244    mut watermarks_rx: watch::Receiver<Arc<Watermarks>>,
245    filter: S::Filter,
246    after: Option<S::Cursor>,
247    after_checkpoint: Option<u64>,
248    subscriber_limit: SubscriberLimit,
249    config: SubscriptionConfig,
250) -> Result<impl Stream<Item = Result<Edge<String, S::Item, EmptyFields>, RpcError>>, RpcError<Error>>
251where
252    for<'a> CursorToken: From<&'a S::Cursor>,
253{
254    // The checkpoint to resume after: the later of the `after` cursor and `afterCheckpoint`. A
255    // cursor's checkpoint is read through its `CursorToken` position, so the driver needs no
256    // per-feed accessor.
257    let start_from = match (
258        after
259            .as_ref()
260            .map(|c| CursorToken::from(c).position.checkpoint()),
261        after_checkpoint,
262    ) {
263        (Some(a), Some(b)) => Some(a.max(b)),
264        (a, b) => a.or(b),
265    };
266
267    // Reject a far-ahead resume before admitting, so bad input never claims a slot or lands in the
268    // subscription lifecycle metrics.
269    reject_if_start_too_far_ahead(start_from, &broadcast, &config)?;
270
271    // Claim a concurrency slot for the subscription's lifetime; at capacity this is `Err` with no
272    // guard created (and so no lifecycle metric).
273    let guard = SubscriptionLifecycleGuard::new(
274        S::subscription_type(),
275        broadcast.metrics(),
276        &subscriber_limit,
277    )
278    .map_err(upcast)?;
279
280    // Size the backfill scan page to the resolve concurrency. Scans are sequential (each needs the
281    // previous page's cursor), so feeding one window of `n` concurrent resolutions takes ceil(n /
282    // page) scans: a page much smaller than the concurrency makes scanning the bottleneck, a much
283    // larger one just holds a bigger page in memory. Matching them is roughly one scan per window.
284    let scan_page_size = config.max_concurrent_resolutions;
285
286    // Pin the handoff once the scan comes within half the live buffer of the tip, leaving room for
287    // checkpoints that arrive during the handoff so the receiver does not lag.
288    let handoff_threshold = config.broadcast_buffer as u64 / 2;
289
290    Ok(stream! {
291        let mut pending_receiver = None;
292        let mut handoff = Handoff::new(handoff_threshold, start_from);
293        if after.as_ref().is_some_and(|c| has_placeholder_checkpoint(&CursorToken::from(c))) {
294            handoff = handoff.resume_from_first_output();
295        }
296        let mut last_checkpoint: Option<u64> = None;
297
298        // Phase 1: scan toward the tip, pinning the live receiver near it. `after` and/or
299        // `afterCheckpoint` resume the backfill; neither means live-only.
300        if let Some(resume_checkpoint) = start_from {
301            // The resume point can be ahead of this server's watermarks (a cursor minted from the
302            // live stream, or by a more caught-up server). Waiting for them to reach it raises the
303            // ledger readers' monotonic-read floor past it, so no replica scans from a view behind
304            // the resume point. `scan_page` then waits for each page's end cursor the same way,
305            // keeping the floor at or above every later resume cursor.
306            if let Err(e) =
307                wait_for_pipelines_catching_up_at(resume_checkpoint, &mut watermarks_rx).await
308            {
309                guard.terminate(SubscriptionTerminationReason::BackfillError);
310                yield Err(e.into());
311                return;
312            }
313
314            let checkpoint_lo = after_checkpoint.map_or(0, |cp| cp.saturating_add(1));
315            let scan = backfill::<S>(
316                reader,
317                caches.clone(),
318                resolver_limits.clone(),
319                watermarks_rx,
320                filter.clone(),
321                after,
322                checkpoint_lo,
323                scan_page_size,
324            );
325            for await scanned in scan {
326                let scanned = match scanned {
327                    Ok(scanned) => scanned,
328                    Err(e) => {
329                        guard.terminate(SubscriptionTerminationReason::BackfillError);
330                        yield Err(e);
331                        return;
332                    }
333                };
334
335                let step = handoff.step(scanned, broadcast.network_tip(), || {
336                    pending_receiver = Some(broadcast.broadcaster().resubscribe());
337                    broadcast.network_tip()
338                });
339                match step {
340                    ScanStep::Deliver(edge) => {
341                        yield Ok(edge);
342                        guard.record_backfill_delivered();
343                    }
344                    ScanStep::Continue => {}
345                    ScanStep::Done => break,
346                }
347            }
348            last_checkpoint = handoff.pinned();
349        }
350
351        // Phase 2: follow live from `handoff + 1` (a fresh receiver if there was no backfill).
352        let receiver = pending_receiver.unwrap_or_else(|| broadcast.broadcaster().resubscribe());
353        for await edge in
354            live::<S>(receiver, last_checkpoint, caches, resolver_limits, filter, guard)
355        {
356            yield edge;
357        }
358    })
359}

361/// Reject a start point sitting more than `max_ahead` checkpoints past the chain tip. There is
362/// nothing to backfill ahead of the tip, so such a request would only wait for the chain to reach
363/// it, and a far-future one would hold the connection open indefinitely.
364fn reject_if_start_too_far_ahead(
365    start_from: Option<u64>,
366    broadcast: &SubscriptionBroadcast,
367    config: &SubscriptionConfig,
368) -> Result<(), RpcError<Error>> {
369    let max_ahead = config.max_start_checkpoints_ahead_of_tip;
370    if let Some(start) = start_from
371        && start > broadcast.network_tip().saturating_add(max_ahead)
372    {
373        return Err(bad_user_input(Error::TooFarAheadOfTip { max: max_ahead }));
374    }
375    Ok(())
376}

378/// Scan matches from the resume point toward the tip, yielding each match and a per-page coverage
379/// marker. Open-ended: the caller stops it once the handoff is covered.
380fn backfill<S: Subscribable>(
381    reader: AlphaLedgerGrpcReader,
382    caches: Arc<StreamedCaches>,
383    resolver_limits: sui_package_resolver::Limits,
384    mut watermarks_rx: watch::Receiver<Arc<Watermarks>>,
385    filter: S::Filter,
386    mut after: Option<S::Cursor>,
387    checkpoint_lo: u64,
388    scan_page_size: usize,
389) -> impl Stream<Item = Result<Scanned<S::Item>, RpcError>>
390where
391    for<'a> CursorToken: From<&'a S::Cursor>,
392{
393    stream! {
394        let limits = PageLimits {
395            default: scan_page_size as u32,
396            max: scan_page_size as u32,
397        };
398
399        // Resume from whichever is later: the `after` cursor or `[checkpoint_lo, ..]`. The upper
400        // bound is `u64::MAX` because backfill is open-ended, chasing the moving tip; `subscribe`
401        // stops it once the handoff is covered.
402        let cp_bounds = checkpoint_lo..=u64::MAX;
403
404        loop {
405            let (items, next_after) = match scan_page::<S>(
406                &reader,
407                &caches,
408                &resolver_limits,
409                &limits,
410                cp_bounds.clone(),
411                after.as_ref(),
412                &filter,
413                &mut watermarks_rx,
414            )
415            .await
416            {
417                Ok(page) => page,
418                Err(e) => {
419                    yield Err(e);
420                    return;
421                }
422            };
423            for item in items {
424                yield Ok(item);
425            }
426            after = Some(next_after);
427        }
428    }
429}

431/// The highest checkpoint fully scanned as of `token`, or `None` if no whole checkpoint is covered.
432///
433/// Only checkpoints below the cursor's are covered: an `Item` cursor's checkpoint may hold later
434/// matches, and a list call that drains ledger tip `T` ends with a `Boundary` cursor at `T + 1`.
435fn covered_checkpoint(token: &CursorToken) -> Option<u64> {
436    token.position.checkpoint().checked_sub(1)
437}

439/// Whether `token` is a transaction or event cursor minted by the Postgres path, which records
440/// checkpoint 0 in place of the item's real checkpoint.
441fn has_placeholder_checkpoint(token: &CursorToken) -> bool {
442    match token.position {
443        Position::Checkpoints { .. } => false,
444        Position::Transactions { checkpoint, tx_seq } => checkpoint == 0 && tx_seq != 0,
445        Position::Events {
446            checkpoint, tx_seq, ..
447        } => checkpoint == 0 && tx_seq != 0,
448    }
449}

451/// Whether a page ending at `end` moved the scan past `after`. A page that drains the ledger's tip
452/// ends where it started, echoing `after`, so the scan waits before asking again. The ledger
453/// rejects an `after` ahead of its indexed tip as out of range rather than answering from its own
454/// tip (#28308), so a page never ends behind `after`.
455fn made_progress(end: &CursorToken, after: Option<&CursorToken>) -> bool {
456    after != Some(end)
457}

459/// Follow the live broadcast from `last_checkpoint + 1`, delivering matching items. Drops the
460/// one-checkpoint seam overlap (the resubscribe/tip race), gap-checks, and disconnects on anomalies.
461fn live<S: Subscribable>(
462    mut receiver: CheckpointBroadcaster,
463    mut last_checkpoint: Option<u64>,
464    caches: Arc<StreamedCaches>,
465    resolver_limits: sui_package_resolver::Limits,
466    filter: S::Filter,
467    guard: SubscriptionLifecycleGuard,
468) -> impl Stream<Item = Result<Edge<String, S::Item, EmptyFields>, RpcError>> {
469    stream! {
470        let mut delivered_live = false;
471        loop {
472            match receiver.recv().await {
473                Ok(checkpoint) => {
474                    let seq = checkpoint.summary.sequence_number;
475                    if let Some(last) = last_checkpoint {
476                        // Already covered by the scan (resubscribe/tip overlap): skip.
477                        if seq <= last {
478                            continue;
479                        }
480                        if seq > last + 1 {
481                            warn!(
482                                last_checkpoint = last,
483                                received = seq,
484                                "Unexpected gap between scan and live; disconnecting"
485                            );
486                            guard.terminate(SubscriptionTerminationReason::UnexpectedGap);
487                            yield Err(reconnect_error());
488                            return;
489                        }
490                    }
491                    // Deliver each matching item as its own payload, ordered within the checkpoint.
492                    // Empty checkpoints yield nothing.
493                    let edges = match S::matching_edges(&checkpoint, &caches, &resolver_limits, &filter) {
494                        Ok(edges) => edges,
495                        Err(e) => {
496                            guard.terminate(SubscriptionTerminationReason::Error);
497                            yield Err(e);
498                            return;
499                        }
500                    };
501                    for edge in edges {
502                        yield Ok(edge);
503                        guard.record_delivered(checkpoint.summary.timestamp_ms);
504                    }
505                    last_checkpoint = Some(seq);
506                    delivered_live = true;
507                }
508                // A lag before the first live checkpoint is catch-up overflow (likely kv-rpc lag).
509                Err(broadcast::error::RecvError::Lagged(missed)) if !delivered_live => {
510                    warn!(missed, "Subscriber fell behind during catch-up; disconnecting");
511                    guard.terminate(SubscriptionTerminationReason::Lagged);
512                    yield Err(reconnect_error());
513                    return;
514                }
515                Err(e) => {
516                    guard.terminate(SubscriptionTerminationReason::from_recv_error(&e));
517                    yield Err(broadcast_error(e));
518                    return;
519                }
520            }
521        }
522    }
523}

525/// Retry policy for the backfill scan: jittered exponential backoff, giving up after 60s (past which
526/// a failure is unlikely to be a transient blip).
527fn scan_backoff() -> ExponentialBackoff {
528    ExponentialBackoff {
529        initial_interval: Duration::from_millis(100),
530        max_interval: Duration::from_secs(5),
531        max_elapsed_time: Some(Duration::from_secs(60)),
532        ..Default::default()
533    }
534}

536/// Paces re-requests while the ledger has nothing past the scan's position: waits double from
537/// `BACKFILL_POLL_INTERVAL` up to `BACKFILL_POLL_MAX_INTERVAL`, and once `BACKFILL_POLL_DEADLINE`
538/// has passed without progress the client is asked to reconnect.
539struct LedgerPoll {
540    interval: Duration,
541    deadline: tokio::time::Instant,
542}
543
544impl LedgerPoll {
545    fn new() -> Self {
546        Self {
547            interval: BACKFILL_POLL_INTERVAL,
548            deadline: tokio::time::Instant::now() + BACKFILL_POLL_DEADLINE,
549        }
550    }
551
552    async fn wait(&mut self) -> Result<(), RpcError> {
553        if tokio::time::Instant::now() >= self.deadline {
554            warn!("Ledger made no progress within the backfill deadline; disconnecting");
555            return Err(reconnect_error());
556        }
557        tokio::time::sleep(self.interval).await;
558        self.interval = (self.interval * 2).min(BACKFILL_POLL_MAX_INTERVAL);
559        Ok(())
560    }
561}

563/// Scan the next page of matches, waiting at the indexer tip if nothing is ready yet. Returns the
564/// matches, a trailing coverage marker, and the cursor to resume from.
565async fn scan_page<S: Subscribable>(
566    reader: &AlphaLedgerGrpcReader,
567    caches: &Arc<StreamedCaches>,
568    resolver_limits: &sui_package_resolver::Limits,
569    limits: &PageLimits,
570    cp_bounds: RangeInclusive<u64>,
571    after: Option<&S::Cursor>,
572    filter: &S::Filter,
573    watermarks_rx: &mut watch::Receiver<Arc<Watermarks>>,
574) -> Result<(Vec<Scanned<S::Item>>, S::Cursor), RpcError>
575where
576    for<'a> CursorToken: From<&'a S::Cursor>,
577{
578    let mut poll = LedgerPoll::new();
579    loop {
580        let result = scan_with_retry::<S>(reader, limits, cp_bounds.clone(), after, filter).await?;
581
582        // No end cursor means nothing past the indexer tip is indexed yet; wait, then re-request. A
583        // range that was scanned but matched nothing still returns a boundary cursor, not `None`.
584        let Some(end_cursor) = result.last_cursor() else {
585            poll.wait().await?;
586            continue;
587        };
588        let end_cursor = CursorToken::decode(end_cursor).context("Failed to decode scan cursor")?;
589
590        // A page that drained the ledger's tip without moving: wait for the ledger to index more
591        // before asking again, rather than re-polling at full speed.
592        if !made_progress(&end_cursor, after.map(CursorToken::from).as_ref()) {
593            poll.wait().await?;
594            continue;
595        }
596
597        // Hold the page until all pipelines have indexed through its end.
598        wait_for_pipelines_catching_up_at(end_cursor.position.checkpoint(), watermarks_rx).await?;
599
600        // A forward page keeps `items` in ascending order, one delivered edge each. The decoded
601        // token yields both the checkpoint and the edge cursor, so it is only decoded once.
602        let mut items = Vec::with_capacity(result.items.len() + 1);
603        for item in &result.items {
604            let token =
605                CursorToken::decode(&item.cursor).context("Failed to decode scan item cursor")?;
606            let checkpoint = token.position.checkpoint();
607            let cursor = S::Cursor::try_from(token)
608                .context("Unexpected position in scan item cursor")?
609                .encode_cursor();
610            items.push(Scanned::Match {
611                checkpoint,
612                edge: Edge::new(
613                    cursor,
614                    S::build_node(caches, resolver_limits, &item.payload)?,
615                ),
616            });
617        }
618        // Coverage marker at the fully-scanned end: advances the handoff even on a match-less page.
619        if let Some(checkpoint) = covered_checkpoint(&end_cursor) {
620            items.push(Scanned::Covered { checkpoint });
621        }
622
623        let next_after =
624            S::Cursor::try_from(end_cursor).context("Unexpected position in scan end cursor")?;
625        return Ok((items, next_after));
626    }
627}

629/// Scan one page over `cp_bounds`, retrying transient failures under a bounded budget.
630async fn scan_with_retry<S: Subscribable>(
631    reader: &AlphaLedgerGrpcReader,
632    limits: &PageLimits,
633    cp_bounds: RangeInclusive<u64>,
634    after: Option<&S::Cursor>,
635    filter: &S::Filter,
636) -> Result<StreamPage<S::ScanItem>, RpcError> {
637    let page = Page::from_params(
638        limits,
639        Some(limits.default as u64),
640        after.cloned(),
641        None,
642        None,
643    )
644    .map_err(anyhow::Error::from)?;
645
646    backoff::future::retry(scan_backoff(), || async {
647        S::scan(reader, cp_bounds.clone(), &page, filter)
648            .await
649            .map_err(|e| {
650                warn!(error = ?e, "scan failed, retrying");
651                backoff::Error::transient(e)
652            })
653    })
654    .await
655}

689    /// Runs scan outputs through [`Handoff::step`] against a fixed streamed `tip`, as the backfill
690    /// loop in `subscribe` does. Returns the delivered match checkpoints and the checkpoint live
691    /// starts after (`None` if still scanning).
692    fn drive(
693        threshold: u64,
694        resume_from: Option<u64>,
695        tip: u64,
696        outputs: &[Out],
697    ) -> (Vec<u64>, Option<u64>) {
698        drive_handoff(Handoff::new(threshold, resume_from), tip, outputs)
699    }
700
701    /// [`drive`] with a prepared `handoff`.
702    fn drive_handoff(mut handoff: Handoff, tip: u64, outputs: &[Out]) -> (Vec<u64>, Option<u64>) {
703        let mut delivered = vec![];
704        for out in outputs {
705            // Matches carry their checkpoint as the node, so delivery order is observable.
706            let scanned = match out {
707                Out::Match(checkpoint) => Scanned::Match {
708                    checkpoint: *checkpoint,
709                    edge: Edge::new(String::new(), *checkpoint),
710                },
711                Out::End(token) => match covered_checkpoint(token) {
712                    Some(checkpoint) => Scanned::Covered { checkpoint },
713                    None => continue,
714                },
715            };
716            match handoff.step(scanned, tip, || tip) {
717                ScanStep::Deliver(edge) => delivered.push(edge.node),
718                ScanStep::Continue => {}
719                ScanStep::Done => return (delivered, handoff.pinned()),
720            }
721        }
722        (delivered, None)
723    }
